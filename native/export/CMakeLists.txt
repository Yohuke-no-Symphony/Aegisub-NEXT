cmake_minimum_required(VERSION 3.25)
if(APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Product macOS deployment target")
endif()
project(AegiNextExport VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MINGW)
    add_link_options(-static)
endif()
if(MSVC)
    add_compile_options(/W4 /WX)
else()
    add_compile_options(-Wall -Wextra -Werror)
endif()
set(AEGINEXT_FFMPEG_ROOT "" CACHE PATH "Pinned FFmpeg SDK")
set(AEGINEXT_NATIVE_OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/lib" CACHE PATH "Native output")
if(NOT AEGINEXT_FFMPEG_ROOT)
    message(FATAL_ERROR "Explicit AEGINEXT_FFMPEG_ROOT is required")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Export ABI requires a 64-bit process")
endif()
unset(FFMPEG_INCLUDE CACHE)
find_path(FFMPEG_INCLUDE libavutil/csp.h PATHS "${AEGINEXT_FFMPEG_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
foreach(COMPONENT avformat avcodec avutil swscale)
    unset(FFMPEG_${COMPONENT} CACHE)
    find_library(FFMPEG_${COMPONENT} NAMES ${COMPONENT} ${COMPONENT}.dll PATHS "${AEGINEXT_FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    list(APPEND FFMPEG_LIBRARIES "${FFMPEG_${COMPONENT}}")
endforeach()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/AegiNext.Media/Probing/ffmpeg-toolchain.json" MANIFEST)
set(VERSION_HEADER "#pragma once\n")
foreach(COMPONENT avformat avcodec avutil swscale)
    string(JSON VERSION GET "${MANIFEST}" libraries "lib${COMPONENT}")
    string(REPLACE "." ";" PARTS "${VERSION}")
    list(GET PARTS 0 MAJOR)
    list(GET PARTS 1 MINOR)
    list(GET PARTS 2 MICRO)
    math(EXPR VERSION_INT "(${MAJOR} << 16) | (${MINOR} << 8) | ${MICRO}")
    string(TOUPPER "${COMPONENT}" UPPER)
    string(APPEND VERSION_HEADER "#define AN_EXPORT_${UPPER} ${VERSION_INT}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/export_versions.h" "${VERSION_HEADER}")
if(NOT TARGET aeginext_media_core)
    set(FFMPEG_INCLUDE_DIR "${FFMPEG_INCLUDE}")
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../shared" "${CMAKE_CURRENT_BINARY_DIR}/shared")
endif()
find_package(Threads REQUIRED)
add_library(aeginext_export SHARED src/export_api.cpp src/encoder_backend.cpp src/qsv_backend.cpp src/export_metadata.cpp src/export_muxer.cpp src/color_pipeline.cpp src/export_performance.cpp src/yuv_frame_pipeline.cpp src/frame_row_executor.cpp src/prepared_overlay.cpp src/decode_prefetcher.cpp)
set(EXPORT_CONTRACT_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/include/aeginext_export.h")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${EXPORT_CONTRACT_HEADER}")
file(SHA256 "${EXPORT_CONTRACT_HEADER}" EXPORT_CONTRACT_HASH)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/export_contract.sha256" "${EXPORT_CONTRACT_HASH}\n")
add_custom_command(OUTPUT "${AEGINEXT_NATIVE_OUTPUT_DIR}/aeginext_export.contract.sha256"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different
        "${CMAKE_CURRENT_BINARY_DIR}/export_contract.sha256"
        "${AEGINEXT_NATIVE_OUTPUT_DIR}/aeginext_export.contract.sha256"
    DEPENDS aeginext_export "$<TARGET_FILE:aeginext_export>" "${CMAKE_CURRENT_BINARY_DIR}/export_contract.sha256"
    VERBATIM)
add_custom_target(aeginext_export_contract ALL
    DEPENDS "${AEGINEXT_NATIVE_OUTPUT_DIR}/aeginext_export.contract.sha256")
target_include_directories(aeginext_export PUBLIC include PRIVATE src "${FFMPEG_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(aeginext_export PRIVATE aeginext_media_core ${FFMPEG_LIBRARIES} Threads::Threads)
target_compile_definitions(aeginext_export PRIVATE AEGINEXT_EXPORT_BUILD)
set_target_properties(aeginext_export PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${AEGINEXT_NATIVE_OUTPUT_DIR}" RUNTIME_OUTPUT_DIRECTORY "${AEGINEXT_NATIVE_OUTPUT_DIR}" CXX_VISIBILITY_PRESET hidden BUILD_RPATH "${AEGINEXT_FFMPEG_ROOT}/lib")
if(WIN32)
    set_target_properties(aeginext_export PROPERTIES PREFIX "")
endif()
